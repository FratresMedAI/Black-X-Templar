from scapy.all import sr1
import time

def send_RST(src_ip, dst_ip, port):
    rst = sr1(IP(src=src_ip, dst=dst_ip)/TCP(sport=0, dport=port, flags='R'), timeout=2)
    return rst is not None  # Returns True if packet was sent

def kill_critical_threat(threat_record):
    src_ip, dst_ip, port = threat_record["source"], threat_record["destination"], threat_record["port"]
    if send_RST(src_ip, dst_ip, port) and send_RST(dst_ip, src_ip, port):
        print(f"Threat severed: {src_ip}:{port} → {dst_ip}")
        return True
    return False
