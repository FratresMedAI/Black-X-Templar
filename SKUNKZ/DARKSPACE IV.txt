# enforcer.py
from scapy.all import sniff, IP, TCP, Raw
import json
import re
import os
import sys

# Load configuration
sys.path.append(os.path.dirname(os.path.abspath(__file__)))
import config

# Function to check if an IP is in any of the authorized agent ranges
def is_authorized(ip):
    for agent in config.AUTHORIZED_AGENTS:
        # Simple check: if the IP matches the subnet, it's authorized
        if ip in agent or re.match(agent, ip):
            return True
    return False

# Function to check if an IP is in any of the LLM endpoint ranges
def is_llm_endpoint(ip):
    for provider, nets in config.LLM_ENDPOINTS.items():
        for net in nets:
            if ip in net:
                return provider
    return None

# Packet callback function
def packet_callback(packet):
    # Only process TCP packets with a payload
    if packet.haslayer(TCP) and packet.haslayer(Raw):
        src_ip = packet[IP].src
        dst_ip = packet[IP].dst
        
        # Check if the destination is an LLM endpoint
        provider = is_llm_endpoint(dst_ip)
        if provider:
            # Check if the source is not authorized
            if not is_authorized(src_ip):
                # Check for HTTP POST in the payload
                payload = packet[TCP].payload
                if b"POST" in payload:

Task 2: app.py (Fratres Forensic Hashing)
We'll update the audit_log function to include a hash.

# app.py
import os
import hmac
import hashlib
import sqlite3
from config import DB_PATH, SECRET_KEY

def audit_log(threat_details):
    # Generate HMAC signature
    signature = hmac.new(SECRET_KEY, str(threat_details).encode('utf-8'), hashlib.sha256).hexdigest()
    
    # Connect to the database
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    
    # Create table if it doesn't exist
    c.execute('''CREATE TABLE IF NOT EXISTS audit_log (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                threat_details TEXT,
                hmac_signature TEXT,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP)''')
    
    # Insert the log entry
    c.execute("INSERT INTO audit_log (threat_details, hmac_signature) VALUES (?, ?)",
              (threat_details, signature))
    conn.commit()
    conn.close()

# Example usage
if __name__ == '__main__':

    audit_log(threat)
Task 3: kill_chain.py (Kinetic Kill-Switch)
This script will monitor the threat_log table for critical threats and send RST packets.

# kill_chain.py
from scapy.all import sr1, TCP, IP
import time
import sqlite3
from config import DB_PATH

def send_RST_packet(src_ip, dst_ip, sport, dport):
    # Send RST packet from src_ip to dst_ip on port dport
    rst = sr1(IP(src=src_ip, dst=dst_ip)/TCP(sport=sport, dport=dport, flags='R'), verbose=0)
    return rst is not None

def kill_critical_threat(threat_record):
    src_ip = threat_record['source']
    dst_ip = threat_record['destination']
    port = threat_record['port']  # We assume port is stored

    # First, send RST from source to destination
    if send_RST_packet(src_ip, dst_ip, sport=0, dport=port):
        print(f"Sent RST from {src_ip}:{port} to {dst_ip}")
    else:
        print(f"Failed to send RST from {src_ip}:{port} to {dst_ip}")

    # Then, send RST from destination to source
    if send_RST_packet(dst_ip, src_ip, sport=0, dport=port):
        print(f"Sent RST from {dst_ip}:{port} to {src_ip}")
    else:
        print(f"Failed to send RST from {dst_ip}:{port} to {src_ip}")

    # Optionally, remove the threat record from the database
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("DELETE FROM threat_log WHERE id=?", (threat_record['id'],))
    conn.commit()
    conn.close()

def monitor_threat_log():
    while True:
        conn = sqlite3.connect(DB_PATH)
        c = conn.cursor()
        c.execute("SELECT * FROM threat_log WHERE threat_type='Critical'")
        critical_threats = c.fetchall()
        conn.close()

        for threat in critical_threats:
            print(f"Critical threat detected: {threat}")
            kill_critical_threat(threat)

        time.sleep(30)  # Check every 30 seconds

if __name__ == '__main__':
    monitor_threat_log()
